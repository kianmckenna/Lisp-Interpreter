(def length (list)
    (if
       (nil? list)
       0
       (add 1 (length (cdr list))) 
    )
)

(def contains? (item list)
    (if
        (nil? list)
        ()
        (if 
            (eq? (car list) item)
            'T
            (contains? item (cdr list))
        )
    )
)

(def count (item list)
    (if
        (nil? list)
        0
        (if
            (eq? item (car list))
            (add 1 (count item (cdr list))) 
            (count item (cdr list))  
        )
    )
)

(def construct (list)
    (if
        (nil? list)
        ()
        (cons (car list) (construct (cdr list)))
    )
)

(def append (list1 list2)
    (foldr (function (elem acc)
        (cons elem acc)
    ) list2 list1)
)

(def allNumbers? (list)
    (if
        (nil? list)
        'T
        (if
            (number? (car list))
            (allNumbers? (cdr list))
            ()
        )
    )
)

(def reverse (list)
    (foldl (function (x y)
                (cons y x)
            ) () list)
)

(def compare (num1 num2)
    (if
        (lt num1 num2)
        num2
        num1
    )
)

(def max (list)
    (if
        (nil? (cdr list))
        (car list)
        (if
            (nil? (cdr list))
            list
            (compare (car list) (max (cdr list)))
        )
    )
)

(def lesser (num1 num2)
    (if
        (lt num1 num2)
        num1
        num2
    )
)

(def min (list)
    (if
        (nil? (cdr list))
        (car list)
        (if
            (nil? (cdr list))
            list
            (lesser (car list) (min (cdr list)))
        )
    )
)

(def map (transform list)
    (foldr (function (elem acc) 
        (cons (transform elem) acc)
    ) () list)
)

(def filter (check list)
    (foldr (function (elem acc)
        (if (check elem)
            (cons elem acc)
            acc
        )
    ) () list)
)

(def foldl (op acc list)
    (if (nil? list)
        acc
        (foldl op (op acc (car list)) (cdr list))
    )
)

(def foldr (op acc list)
    (if (nil? list)
        acc
        (op (car list) (foldr op acc (cdr list)))
    )
)
